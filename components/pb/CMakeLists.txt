set(NANOPB_SRC_ROOT_FOLDER "${COMPONENT_DIR}/nanopb")
set(NANOPB_GENERATOR "${NANOPB_SRC_ROOT_FOLDER}/generator-bin/protoc")
set(ROOT_PROTO_FILES_PATH ${COMPONENT_DIR}/proto)

set(NANOPB_OPTS --nanopb_opt=-I${ROOT_PROTO_FILES_PATH})
set(NANOPB_OUT --nanopb_out=${COMPONENT_DIR})

FILE(GLOB ALL_PROTO_FILES ${ROOT_PROTO_FILES_PATH}/*.proto)
execute_process(COMMAND ${NANOPB_GENERATOR} "${NANOPB_OPTS}" ${NANOPB_OUT} -I=${ROOT_PROTO_FILES_PATH} ${ALL_PROTO_FILES})

FILE(GLOB SRC_FILES *.c *.cpp)

                       INCLUDE_DIRS "." ${NANOPB_SRC_ROOT_FOLDER})
